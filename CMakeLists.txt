cmake_minimum_required(VERSION 3.24)
project(networked_simulation LANGUAGES CXX)
include(FetchContent)
include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FLATBUFFERS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATLIB OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(flatbuffers URL https://codeload.github.com/google/flatbuffers/tar.gz/refs/tags/v25.2.10)
FetchContent_Declare(asio URL https://codeload.github.com/chriskohlhoff/asio/tar.gz/refs/tags/asio-1-30-2)
FetchContent_MakeAvailable(flatbuffers asio)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/wire_generated.h"
    COMMAND flatc --cpp -o "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/wire.fbs"
    DEPENDS flatc wire.fbs VERBATIM)
add_custom_target(protocol DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/wire_generated.h")
foreach(target sim_server sim_load)
    add_executable(${target} ${target}.cpp)
    add_dependencies(${target} protocol)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
        "${asio_SOURCE_DIR}/asio/include" "${flatbuffers_SOURCE_DIR}/include")
    target_compile_definitions(${target} PRIVATE ASIO_STANDALONE)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(WIN32)
        target_compile_definitions(${target} PRIVATE _WIN32_WINNT=0x0A00)
        target_link_libraries(${target} PRIVATE ws2_32 mswsock)
    endif()
endforeach()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME loopback COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/measure.py"
        $<TARGET_FILE:sim_server> $<TARGET_FILE:sim_load> --counts 4 --seconds 2)
    set_tests_properties(loopback PROPERTIES TIMEOUT 15)
endif()
